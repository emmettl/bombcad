"""Periodic read-only ownership/resource observations around one existing-lock profiler run."""
from pathlib import Path
import json,os,subprocess,time,sys
root=Path(__file__).resolve().parent;label=sys.argv[1];jobs=sys.argv[2:];state=root/(label+'-state.json');samples=[]
def probe(args):
 try:
  p=subprocess.run(args,capture_output=True,text=True,timeout=10);return dict(returncode=p.returncode,stdout=p.stdout,stderr=p.stderr)
 except subprocess.TimeoutExpired:return dict(observationTimedOut=True)
def sample():
 lock=Path.home()/'bombcad-ml/lock/owner'
 try:owner=json.loads(lock.read_text())
 except (OSError,ValueError):owner=None
 return dict(rawSeconds=time.clock_gettime(time.CLOCK_MONOTONIC_RAW),owner=owner,gpu=probe(['nvidia-smi','--query-gpu=name,driver_version,memory.used,utilization.gpu','--format=csv']),processes=probe(['nvidia-smi','--query-compute-apps=pid,process_name,used_memory','--format=csv']))
cmd=[str(Path.home()/'bombcad-ml/venv/bin/python'),str(root/'profile-worker.py'),'--baseline',str(root/'baseline.py'),'--candidate',str(root/'candidate.py'),'--source',str(root/'solver.cu'),'--jobs',*[str(root/j) for j in jobs],'--out',str(root/label),'--lock',str(Path.home()/'bombcad-ml/lock'),'--lock-helper',str(root/'remote-train.py'),'--baseline-cache-dir',str(Path.home()/'bombcad-cuda/ptx-cache'),'--candidate-cache-dir',str(Path.home()/'bombcad-cuda/ptx-cache'),'--fresh-process']
samples.append(sample());assert samples[-1]['owner'] is None,'existing GPU lease; no job started'
with (root/(label+'.log')).open('wb') as out:
 p=subprocess.Popen(cmd,stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
 report=dict(status='running',pid=p.pid,pgid=p.pid,command=cmd,samples=samples);state.write_text(json.dumps(report,indent=2)+'\n')
 while p.poll() is None:
  time.sleep(1);samples.append(sample());state.write_text(json.dumps(report,indent=2)+'\n')
 report.update(status='complete' if p.returncode==0 else 'failed',returncode=p.returncode);samples.append(sample());state.write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(dict(label=label,status=report['status'],returncode=p.returncode,samples=len(samples))))
sys.exit(p.returncode)
